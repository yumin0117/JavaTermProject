```sql
-- 기존 데이터 보존. 프로그램을 모두 종료한 상태에서 최초 한 번 실행합니다.
-- 기존 시퀀스가 있으면 변경하지 않습니다. MAX는 최초 초기화에만 사용합니다.
SET SERVEROUTPUT ON;
DECLARE
  v_exists NUMBER;
  v_start NUMBER;
BEGIN
  SELECT COUNT(*) INTO v_exists FROM user_sequences WHERE sequence_name='SEQ_PLAYLIST';
  IF v_exists=0 THEN
    SELECT GREATEST(NVL(MAX(playlist_id),0)+1,1) INTO v_start FROM PLAYLIST;
    EXECUTE IMMEDIATE 'CREATE SEQUENCE SEQ_PLAYLIST START WITH ' || TO_CHAR(v_start,'FM99999999990') || ' INCREMENT BY 1 MAXVALUE 9999999999 NOCYCLE NOCACHE';
    DBMS_OUTPUT.PUT_LINE('SEQ_PLAYLIST 생성 완료');
  ELSE
    DBMS_OUTPUT.PUT_LINE('기존 SEQ_PLAYLIST 유지');
  END IF;
END;
```

```java
package kr.ac.yumin.playlist.config;

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.nio.file.*;
import java.sql.*;
import java.util.Properties;

/** 접속 정보는 실행 폴더의 외부 파일에서만 읽는다. */
public final class Database {
    private Database() {}

    public static Connection open() throws SQLException {
        Path path = Path.of(System.getProperty("playlist.config", "config/db.properties"));
        Properties p = new Properties();
        try (Reader in = Files.newBufferedReader(path, StandardCharsets.UTF_8)) {
            p.load(in);
        } catch (IOException e) {
            throw new SQLException("config/db.properties를 만들고 Oracle 접속 정보를 입력해 주세요.");
        }
        for (String key : new String[] {"db.url", "db.username", "db.password"})
            if (p.getProperty(key, "").isBlank())
                throw new SQLException("DB 설정의 URL, 사용자 이름, 비밀번호를 모두 입력해 주세요.");
        Properties info = new Properties();
        info.setProperty("user", p.getProperty("db.username"));
        info.setProperty("password", p.getProperty("db.password"));
        info.setProperty("oracle.net.CONNECT_TIMEOUT", "7000");
        info.setProperty("oracle.jdbc.ReadTimeout", "15000");
        return DriverManager.getConnection(p.getProperty("db.url"), info);
    }
}
```

```java
package kr.ac.yumin.playlist.dao;

import java.sql.*;
import java.util.*;

/** 한 연결에서 실행하며 모든 JDBC 자원을 try-with-resources로 닫는다. */
public final class Jdbc {
    private Jdbc() {}

    @FunctionalInterface
    public interface Mapper<T> {
        T read(ResultSet r) throws SQLException;
    }

    public static PreparedStatement prepare(Connection c, String sql, Object... args)
            throws SQLException {
        PreparedStatement p = c.prepareStatement(sql);
        try {
            p.setQueryTimeout(20);
            for (int i = 0; i < args.length; i++) p.setObject(i + 1, args[i]);
            return p;
        } catch (SQLException e) {
            p.close();
            throw e;
        }
    }

    public static int update(Connection c, String sql, Object... args) throws SQLException {
        try (PreparedStatement p = prepare(c, sql, args)) {
            return p.executeUpdate();
        }
    }

    public static <T> List<T> list(Connection c, String sql, Mapper<T> mapper, Object... args)
            throws SQLException {
        try (PreparedStatement p = prepare(c, sql, args);
                ResultSet r = p.executeQuery()) {
            List<T> out = new ArrayList<>();
            while (r.next()) out.add(mapper.read(r));
            return out;
        }
    }

    public static long number(Connection c, String sql, Object... args) throws SQLException {
        return list(c, sql, r -> r.getLong(1), args).get(0);
    }
}
```

```java
package kr.ac.yumin.playlist.dao;

import kr.ac.yumin.playlist.model.*;

import java.sql.*;
import java.util.*;

public final class PlaylistDao {
    private static final String SELECT =
            """
SELECT p.playlist_id,p.title,p.description,p.is_public,p.member_id,m.nickname,p.created_at,
  (SELECT COUNT(*) FROM PLAYLIST_LIKE l WHERE l.playlist_id=p.playlist_id) likes,
  (SELECT COUNT(*) FROM PLAYLIST_LIKE l WHERE l.playlist_id=p.playlist_id AND l.member_id=?) liked
FROM PLAYLIST p JOIN MEMBER m ON m.member_id=p.member_id
""";

    private Playlist map(ResultSet r) throws SQLException {
        return new Playlist(
                r.getLong(1),
                r.getString(2),
                r.getString(3),
                "Y".equals(r.getString(4)),
                r.getString(5),
                r.getString(6),
                r.getTimestamp(7).toLocalDateTime(),
                r.getInt(8),
                r.getInt(9) > 0);
    }

    public List<Playlist> mine(Connection c, String user) throws SQLException {
        return Jdbc.list(
                c,
                SELECT + " WHERE p.member_id=? ORDER BY p.created_at DESC,p.playlist_id DESC",
                this::map,
                user,
                user);
    }

    public List<Playlist> publicLists(Connection c, String user, String query) throws SQLException {
        return Jdbc.list(
                c,
                SELECT
                        + " WHERE p.is_public='Y' AND INSTR(LOWER(p.title),LOWER(?))>0 ORDER BY"
                        + " p.created_at DESC,p.playlist_id DESC",
                this::map,
                user,
                query);
    }

    public List<Playlist> allPublic(Connection c, String user) throws SQLException {
        return Jdbc.list(
                c,
                SELECT + " WHERE p.is_public='Y' ORDER BY p.created_at DESC,p.playlist_id DESC",
                this::map,
                user);
    }

    public Optional<Playlist> visible(Connection c, String user, long id) throws SQLException {
        return Jdbc.list(
                        c,
                        SELECT + " WHERE p.playlist_id=? AND (p.member_id=? OR p.is_public='Y')",
                        this::map,
                        user,
                        id,
                        user)
                .stream()
                .findFirst();
    }

    /** 잠금 후 소유자를 다시 검사한다. 모든 쓰기 작업이 같은 부모 행을 잠근다. */
    public boolean lockOwner(Connection c, String user, long id) throws SQLException {
        return !Jdbc.list(
                        c,
                        "SELECT playlist_id FROM PLAYLIST WHERE playlist_id=? AND member_id=? FOR"
                            + " UPDATE WAIT 5",
                        r -> r.getLong(1),
                        id,
                        user)
                .isEmpty();
    }

    public boolean lockPublic(Connection c, long id) throws SQLException {
        return !Jdbc.list(
                        c,
                        "SELECT playlist_id FROM PLAYLIST WHERE playlist_id=? AND is_public='Y' FOR"
                            + " UPDATE WAIT 5",
                        r -> r.getLong(1),
                        id)
                .isEmpty();
    }

    public long create(Connection c, String user, String title, String description, boolean pub)
            throws SQLException {
        long id = Jdbc.number(c, "SELECT SEQ_PLAYLIST.NEXTVAL FROM DUAL");
        Jdbc.update(
                c,
                "INSERT INTO PLAYLIST(playlist_id,title,description,is_public,member_id)"
                    + " VALUES(?,?,?,?,?)",
                id,
                title,
                description,
                pub ? "Y" : "N",
                user);
        return id;
    }

    public int edit(Connection c, String user, long id, String title, String desc, boolean pub)
            throws SQLException {
        return Jdbc.update(
                c,
                "UPDATE PLAYLIST SET title=?,description=?,is_public=? WHERE playlist_id=? AND"
                    + " member_id=?",
                title,
                desc,
                pub ? "Y" : "N",
                id,
                user);
    }

    public int delete(Connection c, String user, long id) throws SQLException {
        return Jdbc.update(c, "DELETE FROM PLAYLIST WHERE playlist_id=? AND member_id=?", id, user);
    }

    public void like(Connection c, String user, long id, boolean value) throws SQLException {
        if (value)
            Jdbc.update(
                    c,
                    """
INSERT INTO PLAYLIST_LIKE(member_id,playlist_id)
SELECT ?,playlist_id FROM PLAYLIST p WHERE p.playlist_id=? AND p.is_public='Y'
AND NOT EXISTS(SELECT 1 FROM PLAYLIST_LIKE l WHERE l.member_id=? AND l.playlist_id=p.playlist_id)
""",
                    user,
                    id,
                    user);
        else
            Jdbc.update(
                    c, "DELETE FROM PLAYLIST_LIKE WHERE member_id=? AND playlist_id=?", user, id);
    }
}
```

```java
package kr.ac.yumin.playlist.dao;

import kr.ac.yumin.playlist.model.Song;

import java.sql.*;
import java.util.*;

public final class SongDao {
    // 검색 조건과 참여자 집계를 분리하여, 아티스트 한 명으로 검색해도 모든 참여자를 표시한다.
    private static final String COLUMNS =
            """
SELECT s.song_id,s.title,
  (SELECT LISTAGG(a.artist_name,', ' ON OVERFLOW TRUNCATE '...' WITHOUT COUNT)
   WITHIN GROUP(ORDER BY a.artist_id) FROM SONG_ARTIST sa JOIN ARTIST a ON a.artist_id=sa.artist_id
   WHERE sa.song_id=s.song_id) artists,s.album_title,s.genre,s.duration_seconds,
""";

    private Song map(ResultSet r) throws SQLException {
        return new Song(
                r.getLong(1),
                r.getString(2),
                r.getString(3),
                r.getString(4),
                r.getString(5),
                r.getInt(6),
                r.getInt(7));
    }

    public List<Song> search(Connection c, String query) throws SQLException {
        if (query.isEmpty())
            return Jdbc.list(c, COLUMNS + " 0 FROM SONG s ORDER BY s.title,s.song_id", this::map);
        return Jdbc.list(
                c,
                COLUMNS
                        + """
                          0 FROM SONG s WHERE INSTR(LOWER(s.title),LOWER(?))>0 OR EXISTS(
                            SELECT 1 FROM SONG_ARTIST sa JOIN ARTIST a ON a.artist_id=sa.artist_id
                            WHERE sa.song_id=s.song_id AND INSTR(LOWER(a.artist_name),LOWER(?))>0)
                          ORDER BY s.title,s.song_id
                          """,
                this::map,
                query,
                query);
    }

    public List<Song> inPlaylist(Connection c, String user, long id) throws SQLException {
        return Jdbc.list(
                c,
                COLUMNS
                        + """
ps.play_order FROM PLAYLIST_SONG ps JOIN SONG s ON s.song_id=ps.song_id
JOIN PLAYLIST p ON p.playlist_id=ps.playlist_id
WHERE ps.playlist_id=? AND (p.member_id=? OR p.is_public='Y') ORDER BY ps.play_order
""",
                this::map,
                id,
                user);
    }

    public void add(Connection c, String user, long id, long song, int order) throws SQLException {
        Jdbc.update(
                c,
                """
                INSERT INTO PLAYLIST_SONG(playlist_id,song_id,play_order)
                SELECT playlist_id,?,? FROM PLAYLIST WHERE playlist_id=? AND member_id=?
                """,
                song,
                order,
                id,
                user);
    }

    public int remove(Connection c, String user, long id, long song) throws SQLException {
        return Jdbc.update(
                c,
                """
DELETE FROM PLAYLIST_SONG ps WHERE playlist_id=? AND song_id=?
AND EXISTS(SELECT 1 FROM PLAYLIST p WHERE p.playlist_id=ps.playlist_id AND p.member_id=?)
""",
                id,
                song,
                user);
    }

    public void order(Connection c, String user, long id, long song, int order)
            throws SQLException {
        int n =
                Jdbc.update(
                        c,
                        """
UPDATE PLAYLIST_SONG ps SET play_order=? WHERE playlist_id=? AND song_id=?
AND EXISTS(SELECT 1 FROM PLAYLIST p WHERE p.playlist_id=ps.playlist_id AND p.member_id=?)
""",
                        order,
                        id,
                        song,
                        user);
        if (n != 1) throw new SQLException("대상 곡이 변경되었습니다. 새로고침 후 다시 시도해 주세요.");
    }
}
```

```java
package kr.ac.yumin.playlist.security;

import java.security.*;
import java.util.*;

import javax.crypto.SecretKeyFactory;
import javax.crypto.spec.PBEKeySpec;

public final class PasswordHasher {
    private static final int ITERATIONS = 600000;

    private PasswordHasher() {}

    public static String hash(char[] password) {
        byte[] salt = new byte[16];
        new SecureRandom().nextBytes(salt);
        return "pbkdf2_sha256$"
                + ITERATIONS
                + "$"
                + Base64.getEncoder().encodeToString(salt)
                + "$"
                + Base64.getEncoder().encodeToString(derive(password, salt, ITERATIONS));
    }

    public static boolean verify(char[] password, String encoded) {
        if (encoded == null) return false;
        try {
            String[] parts = encoded.split("\\$");
            if (parts.length != 4 || !parts[0].equals("pbkdf2_sha256")) return false;
            int count = Integer.parseInt(parts[1]);
            if (count < 100000 || count > 2000000) return false;
            byte[] salt = Base64.getDecoder().decode(parts[2]),
                    expected = Base64.getDecoder().decode(parts[3]);
            return salt.length == 16
                    && expected.length == 32
                    && MessageDigest.isEqual(expected, derive(password, salt, count));
        } catch (IllegalArgumentException e) {
            return false;
        }
    }

    private static byte[] derive(char[] password, byte[] salt, int count) {
        PBEKeySpec spec = new PBEKeySpec(password, salt, count, 256);
        try {
            return SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256")
                    .generateSecret(spec)
                    .getEncoded();
        } catch (GeneralSecurityException e) {
            throw new IllegalStateException("비밀번호 암호화 기능을 사용할 수 없습니다.", e);
        } finally {
            spec.clearPassword();
        }
    }
}
```

```java
package kr.ac.yumin.playlist.service;

import kr.ac.yumin.playlist.config.Database;
import kr.ac.yumin.playlist.dao.MemberDao;
import kr.ac.yumin.playlist.model.Member;
import kr.ac.yumin.playlist.security.PasswordHasher;

import java.sql.*;
import java.util.Arrays;

public final class AuthService {
    private final MemberDao dao = new MemberDao();

    public static String text(String s, String label, int max, boolean required) {
        String v = s == null ? "" : s.strip();
        if (required && v.isEmpty()) throw new AppException(label + "을(를) 입력해 주세요.");
        if (v.codePointCount(0, v.length()) > max)
            throw new AppException(label + "은(는) " + max + "자 이하여야 합니다.");
        return v;
    }

    public boolean available(String id) throws SQLException {
        id = text(id, "아이디", 30, true);
        try (Connection c = Database.open()) {
            return dao.find(c, id).isEmpty();
        }
    }

    public void register(String id, char[] password, char[] confirm, String nickname, String email)
            throws SQLException {
        try {
            id = text(id, "아이디", 30, true);
            nickname = text(nickname, "닉네임", 50, true);
            email = text(email, "이메일", 254, true);
            if (password.length == 0) throw new AppException("비밀번호를 입력해 주세요.");
            if (password.length > 1024) throw new AppException("비밀번호는 1024자 이하여야 합니다.");
            if (!Arrays.equals(password, confirm)) throw new AppException("비밀번호 확인이 일치하지 않습니다.");
            String hash = PasswordHasher.hash(password);
            try (Connection c = Database.open()) {
                if (dao.find(c, id).isPresent()) throw new AppException("이미 사용 중인 아이디입니다.");
                dao.insert(c, new Member(id, nickname, email), hash);
            } catch (SQLException e) {
                if (e.getErrorCode() == 1) throw new AppException("이미 사용 중인 아이디입니다.");
                throw e;
            }
        } finally {
            Arrays.fill(password, '\0');
            Arrays.fill(confirm, '\0');
        }
    }

    public Member login(String id, char[] password) throws SQLException {
        try {
            id = text(id, "아이디", 30, true);
            try (Connection c = Database.open()) {
                var found = dao.find(c, id);
                if (found.isEmpty() || !PasswordHasher.verify(password, found.get().hash()))
                    throw new AppException("아이디 또는 비밀번호가 올바르지 않습니다. 예시 계정 대신 새로 회원가입해 주세요.");
                return found.get().member();
            }
        } finally {
            Arrays.fill(password, '\0');
        }
    }
}
```

```java
    private final Member member;
    private final PlaylistDao playlists = new PlaylistDao();
    private final SongDao songs = new SongDao();

    public PlaylistService(Member member) {
        this.member = Objects.requireNonNull(member);
    }

    public Member member() {
        return member;
    }

    @FunctionalInterface
    private interface Work<T> {
        T run(Connection c) throws SQLException;
    }

    private <T> T transaction(Work<T> action) throws SQLException {
        try (Connection c = Database.open()) {
            c.setAutoCommit(false);
            try {
                T value = action.run(c);
                c.commit();
                return value;
            } catch (SQLException | RuntimeException e) {
                try {
                    c.rollback();
                } catch (SQLException rollback) {
                    e.addSuppressed(rollback);
                }
                throw e;
            }
        }
    }

    private void owner(Connection c, long id) throws SQLException {
        if (!playlists.lockOwner(c, member.id(), id))
            throw new AppException("본인이 만든 플레이리스트만 변경할 수 있습니다.");
    }

    public List<Playlist> mine() throws SQLException {
        try (Connection c = Database.open()) {
            return playlists.mine(c, member.id());
        }
    }

    public List<Playlist> discover(String query) throws SQLException {
        String q = AuthService.text(query, "검색어", 150, false);
        try (Connection c = Database.open()) {
            return q.isEmpty()
                    ? playlists.allPublic(c, member.id())
                    : playlists.publicLists(c, member.id(), q);
        }
    }

    public List<Song> search(String query) throws SQLException {
        String q = AuthService.text(query, "검색어", 200, false);
        try (Connection c = Database.open()) {
            return songs.search(c, q);
        }
    }

    public PlaylistDetail detail(long id) throws SQLException {
        try (Connection c = Database.open()) {
            // 두 SELECT가 동일 시점의 공개 여부와 수록곡을 보도록 읽기 전용 트랜잭션 사용.
            c.setAutoCommit(false);
            try {
                Jdbc.update(c, "SET TRANSACTION READ ONLY");
                Playlist p =
                        playlists
                                .visible(c, member.id(), id)
                                .orElseThrow(() -> new AppException("조회할 수 없는 플레이리스트입니다."));
                var list = songs.inPlaylist(c, member.id(), id);
                c.commit();
                return new PlaylistDetail(p, List.copyOf(list));
            } catch (SQLException | RuntimeException e) {
                c.rollback();
                throw e;
            }
        }
    }

    public long create(String title, String description, boolean pub) throws SQLException {
        String t = AuthService.text(title, "제목", 150, true),
                d = AuthService.text(description, "설명", 1000, false);
        return transaction(c -> playlists.create(c, member.id(), t, d, pub));
    }

    public void edit(long id, String title, String description, boolean pub) throws SQLException {
        String t = AuthService.text(title, "제목", 150, true),
                d = AuthService.text(description, "설명", 1000, false);
        transaction(
                c -> {
                    owner(c, id);
                    playlists.edit(c, member.id(), id, t, d, pub);
                    return null;
                });
    }

    public void delete(long id) throws SQLException {
        transaction(
                c -> {
                    owner(c, id);
                    playlists.delete(c, member.id(), id);
                    return null;
                });
    }

    public void add(long id, long song) throws SQLException {
        transaction(
                c -> {
                    owner(c, id);
                    var list = songs.inPlaylist(c, member.id(), id);
                    if (list.stream().anyMatch(s -> s.id() == song))
                        throw new AppException("이미 추가한 곡입니다.");
                    int next = list.stream().mapToInt(Song::order).max().orElse(0) + 1;
                    if (next > 999999) throw new AppException("목록에 더 이상 곡을 추가할 수 없습니다.");
                    songs.add(c, member.id(), id, song, next);
                    return null;
                });
    }

    public void remove(long id, long song) throws SQLException {
        transaction(
                c -> {
                    owner(c, id);
                    if (songs.remove(c, member.id(), id, song) != 1)
                        throw new AppException("이미 제거된 곡입니다.");
                    var remaining = songs.inPlaylist(c, member.id(), id);
                    // 오름차순으로 빈 번호에 이동하므로 UNIQUE 위반 없이 1부터 정리된다.
                    for (int i = 0; i < remaining.size(); i++)
                        if (remaining.get(i).order() != i + 1)
                            songs.order(c, member.id(), id, remaining.get(i).id(), i + 1);
                    return null;
                });
    }

    public void move(long id, long song, int direction) throws SQLException {
        if (direction != 1 && direction != -1) throw new AppException("이동 방향이 올바르지 않습니다.");
        transaction(
                c -> {
                    owner(c, id);
                    var list = songs.inPlaylist(c, member.id(), id);
                    int index = -1;
                    for (int i = 0; i < list.size(); i++) if (list.get(i).id() == song) index = i;
                    if (index < 0) throw new AppException("목록에 없는 곡입니다.");
                    int other = index + direction;
                    if (other < 0 || other >= list.size()) return null;
                    Set<Integer> occupied = new HashSet<>();
                    for (Song s : list) occupied.add(s.order());
                    int temp = 1;
                    while (occupied.contains(temp) && temp <= 999999) temp++;
                    if (temp > 999999)
                        throw new AppException("순서를 바꿀 빈 번호가 없습니다. 곡 하나를 제거한 후 다시 시도해 주세요.");
                    Song a = list.get(index), b = list.get(other);
                    songs.order(c, member.id(), id, a.id(), temp);
                    songs.order(c, member.id(), id, b.id(), a.order());
                    songs.order(c, member.id(), id, a.id(), b.order());
                    return null;
                });
    }

    public void like(long id, boolean value) throws SQLException {
        transaction(
                c -> {
                    if (value && !playlists.lockPublic(c, id))
                        throw new AppException("공개 플레이리스트에만 좋아요를 누를 수 있습니다.");
                    playlists.like(c, member.id(), id, value);
                    return null;
                });
    }
```

```java
    public static <T> void work(Component owner, Callable<T> job, Consumer<T> success) {
        JRootPane root = SwingUtilities.getRootPane(owner);
        if (root != null && Boolean.TRUE.equals(root.getClientProperty("busy"))) return;
        if (root != null) {
            root.putClientProperty("busy", true);
            JPanel glass = new JPanel();
            glass.setOpaque(false);
            glass.addMouseListener(new MouseAdapter() {});
            glass.setFocusTraversalKeysEnabled(false);
            glass.addKeyListener(new KeyAdapter() {});
            root.setGlassPane(glass);
            glass.setCursor(Cursor.getPredefinedCursor(Cursor.WAIT_CURSOR));
            glass.setVisible(true);
            glass.setFocusable(true);
            glass.requestFocusInWindow();
        }
        new SwingWorker<T, Void>() {
            @Override
            protected T doInBackground() throws Exception {
                return job.call();
            }

            @Override
            protected void done() {
                if (root != null) {
                    root.getGlassPane().setVisible(false);
                    root.putClientProperty("busy", false);
                }
                try {
                    T result = get();
                    if (owner.isDisplayable()) success.accept(result);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                } catch (ExecutionException | RuntimeException e) {
                    if (owner.isDisplayable()) error(owner, e);
                }
            }
        }.execute();
    }

    public static String safe(String value) {
        return value == null ? "" : value;
    }
```

```java
    private static List<String> baseline(Connection c) throws SQLException {
        List<String> rows = new ArrayList<>();
        for (String table :
                List.of(
                        "MEMBER",
                        "ARTIST",
                        "SONG",
                        "PLAYLIST",
                        "SONG_ARTIST",
                        "PLAYLIST_SONG",
                        "PLAYLIST_LIKE"))
            rows.addAll(
                    Jdbc.list(
                            c,
                            "SELECT * FROM " + table,
                            r -> {
                                StringBuilder s = new StringBuilder(table);
                                for (int i = 1; i <= r.getMetaData().getColumnCount(); i++)
                                    s.append('|').append(r.getString(i));
                                return s.toString();
                            }));
        Collections.sort(rows);
        return rows;
    }
```
